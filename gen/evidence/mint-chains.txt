== chain: classical (root=EC int=EC leaf=EC) ==
$ openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out chains/classical/root.key
$ openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out chains/classical/int.key
$ openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out chains/classical/leaf.key
$ openssl req -x509 -new -key chains/classical/root.key -out chains/classical/root.crt -subj /CN=Selection classical Root -days 90 -addext basicConstraints=critical,CA:TRUE -addext keyUsage=critical,keyCertSign,cRLSign
$ openssl req -new -key chains/classical/int.key -out chains/classical/int.csr -subj /CN=Selection classical Intermediate
$ openssl x509 -req -in chains/classical/int.csr -CA chains/classical/root.crt -CAkey chains/classical/root.key -out chains/classical/int.crt -days 60 -extfile /dev/fd/63
Certificate request self-signature ok
subject=CN=Selection classical Intermediate
$ openssl req -new -key chains/classical/leaf.key -out chains/classical/leaf.csr -subj /CN=localhost
$ openssl x509 -req -in chains/classical/leaf.csr -CA chains/classical/int.crt -CAkey chains/classical/int.key -out chains/classical/leaf.crt -days 30 -extfile /dev/fd/63
Certificate request self-signature ok
subject=CN=localhost
-- verify (gate: a later failure must be selection, not a bad chain) --
$ openssl verify -CAfile chains/classical/root.crt -untrusted chains/classical/int.crt chains/classical/leaf.crt
chains/classical/leaf.crt: OK
